def plusMinus(arr):
    n=len(arr)
    pos=0
    neg=0
    zero=0
    for i in range(n):
        if arr[i]<0:
            neg=neg+1
        if arr[i]>0:
            pos=pos+1
        if arr[i]==0:
            zero=zero+1
    print(f"{(pos/n):.6f}")
    print(f"{(neg/n):.6f}")
    print(f"{(zero/n):.6f}")

if __name__ == '__main__':
    n = int(input().strip())

    arr = list(map(int, input().rstrip().split()))

    plusMinus(arr)
